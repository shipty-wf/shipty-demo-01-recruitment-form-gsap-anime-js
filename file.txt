    function validateStep(stepIndex) {
      let valid = true;

      if (stepIndex === 0) {
        const nameVal = document.getElementById('field-name').value.trim();
        const emailVal = document.getElementById('field-email').value.trim();
        const ghVal = document.getElementById('field-github').value.trim();
        const emailRegex = /^[^\\s@]+@[^\\s@]+\\.[^\\s@]+$/;
        const ghRegex = /^https?:\\/\\/github\\.com\\/[A-Za-z0-9_.-]+\\/?$/;

        if (nameVal.length < 2) {
          document.getElementById('err-name').classList.remove('hidden');
          valid = false;
        } else {
          document.getElementById('err-name').classList.add('hidden');
          state.data.name = nameVal;
        }

        if (!emailRegex.test(emailVal)) {
          document.getElementById('err-email').classList.remove('hidden');
          valid = false;
        } else {
          document.getElementById('err-email').classList.add('hidden');
          state.data.email = emailVal;
        }

        if (!ghRegex.test(ghVal)) {
          document.getElementById('err-github').classList.remove('hidden');
          valid = false;
        } else {
          document.getElementById('err-github').classList.add('hidden');
          state.data.github = ghVal;
        }

        if (!state.data.role) {
          document.getElementById('err-role').classList.remove('hidden');
          valid = false;
        } else {
          document.getElementById('err-role').classList.add('hidden');
        }
      }

      if (stepIndex === 1) {
        if (state.data.skills.length < 2) {
          document.getElementById('err-skills').classList.remove('hidden');
          valid = false;
        } else {
          document.getElementById('err-skills').classList.add('hidden');
        }
      }

      if (stepIndex === 2) {
        const mindsetVal = document.getElementById('field-mindset').value.trim();
        if (mindsetVal.length < 60) {
          document.getElementById('err-mindset').classList.remove('hidden');
          valid = false;
        } else {
          document.getElementById('err-mindset').classList.add('hidden');
          state.data.mindset = mindsetVal;
        }
      }

      if (stepIndex === 3) {
        const agreed = document.getElementById('field-agree').checked;
        if (!agreed) {
          document.getElementById('err-agree').classList.remove('hidden');
          valid = false;
        } else {
          document.getElementById('err-agree').classList.add('hidden');
          state.data.agreed = true;
        }
      }

      return valid;
    }
